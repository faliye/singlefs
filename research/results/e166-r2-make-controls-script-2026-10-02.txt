#!/usr/bin/env python3
# PC-d：从旧产物做三份拷贝 alpha（不改）、beta（改两处）、gamma（改一处）。登记第五节 5.2。
import sys

sys.path.insert(0, "/tmp/claude-1000/e166-r2-run")
import compare as comparer

old_path, scratch = sys.argv[1], sys.argv[2]
with open(old_path, encoding="utf-8") as handle:
    original_lines = handle.read().split("\n")


def replace_field(line, key, new_value):
    fields = comparer.parse_fields(line)
    rebuilt = " ".join("%s=%s" % (field_key, new_value if field_key == key else field_value) for field_key, field_value in fields)
    return "E7RESULT " + rebuilt


alpha = list(original_lines)
beta = list(original_lines)
gamma = list(original_lines)

beta_changes = []
for index, line in enumerate(beta):
    if line.startswith("E7RESULT "):
        fields = comparer.parse_fields(line)
        if comparer.line_name(fields) == "verdict" and comparer.tag_value(fields, "question") == "Q-1":
            beta[index] = replace_field(line, "judgement", "pc_flipped")
            beta_changes.append((index + 1, "judgement", comparer.tag_value(fields, "judgement"), "pc_flipped"))
            break
for index, line in enumerate(beta):
    if line.startswith("E7RESULT "):
        fields = comparer.parse_fields(line)
        if comparer.line_name(fields) == "verdict" and comparer.tag_value(fields, "question") == "Q-4":
            old_value = int(comparer.tag_value(fields, "duplicate_total"))
            beta[index] = replace_field(line, "duplicate_total", str(3 * old_value + 1))
            beta_changes.append((index + 1, "duplicate_total", str(old_value), str(3 * old_value + 1)))
            break

gamma_changes = []
for index, line in enumerate(gamma):
    if not line.startswith("E7RESULT "):
        continue
    fields = comparer.parse_fields(line)
    chosen = None
    for key, value in fields:
        if key in comparer.derived_field_names(fields) and value.isdigit() and int(value) > 0:
            chosen = (key, int(value))
            break
    if chosen is not None:
        key, value = chosen
        if value % 2 != 0:
            print("作废 9：gamma 要改的值不是偶数：行 %d 字段 %s 值 %d" % (index + 1, key, value))
            sys.exit(3)
        gamma[index] = replace_field(line, key, str(value * 3 // 2))
        gamma_changes.append((index + 1, key, str(value), str(value * 3 // 2), comparer.line_name(fields)))
        break

for name, lines in (("alpha", alpha), ("beta", beta), ("gamma", gamma)):
    with open("%s/pc-%s.out" % (scratch, name), "w", encoding="utf-8") as handle:
        handle.write("\n".join(lines))
print("beta 改了：", beta_changes)
print("gamma 改了：", gamma_changes)
