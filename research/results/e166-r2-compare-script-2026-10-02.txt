#!/usr/bin/env python3
# E166 第 2 次重跑登记第六节 Q-11 至 Q-13 的比对器，也是第五节 5.2 PC-d 的比对器（同一份脚本、同一套规则）。
# 用法：compare.py <旧产物> <新产物>。只读两份文件，不读别的。
import re
import sys

FIELD = re.compile(r'(\S+?)=("[^"]*"|\S*)')
CONCLUSION_NAMES = ("verdict", "geometry_sensitivity", "discrimination_selfcheck")
CONCLUSION_TAGS = ("question", "failure", "arm", "stage", "group")
FULL_LINE_TAGS = ("id", "question", "failure", "arm", "stage", "shape", "initial", "transactions", "devices", "group")

# 第七节 7.1、7.2 的新值（登记里写的，不取任何一次产物）
IDENTIFIERS_ISSUED = ",".join(str(number) for number in range(65536, 65548))
SECTION_SEVEN_ANCHORS = {
    "甲6 发树 ID 的起点（0 = 没发）": "65536,0,0",
    "甲6 期末树 ID 水位": "65548",
    "甲6 第一个事务发的树 ID": IDENTIFIERS_ISSUED,
    "补1 A字面 锁外 file=false n=2": "schedules=20 X=18/18 T=216/18",
    "补1 A字面 锁外 file=false n=3": "schedules=1680 X=2970/1674 T=34992/1620",
}
for control_name in ("A去锁", "O整不", "O读不", "P区同"):
    SECTION_SEVEN_ANCHORS["乙2 %s n=2" % control_name] = "T=48/4"
    SECTION_SEVEN_ANCHORS["乙2 %s n=3" % control_name] = "T=1296/72"


def read_result_lines(path):
    with open(path, encoding="utf-8") as handle:
        lines = [line.rstrip("\n") for line in handle]
    return [line for line in lines if line.startswith("E7RESULT ")]  # 不以 E7RESULT 开头的（含 E7INPUT 行）不算


def parse_fields(line):
    body = line[len("E7RESULT "):]
    fields = FIELD.findall(body)
    assert " ".join("%s=%s" % pair for pair in fields) == body, "行切不回原样：" + line[:120]
    return fields


def tag_value(fields, tag):
    for key, value in fields:
        if key == tag:
            return value
    return None


def line_name(fields):
    return tag_value(fields, "name")


def derived_field_names(fields):
    name = line_name(fields)
    if name in ("cell", "control_cell"):
        return {"T_total", "T_max_single"}
    if name == "trajectory":
        return {"T_peak"}
    if name == "verdict" and tag_value(fields, "question") == "Q-4":
        return {"duplicate_total"}
    if name == "discrimination_selfcheck" and (tag_value(fields, "group") or "").endswith(":Q-4"):
        return {"default_value", "sample_value"}
    return set()


def pairing_key(fields, tags):
    return tuple([line_name(fields)] + [(tag, tag_value(fields, tag)) for tag in tags if tag_value(fields, tag) is not None])


def ratio_is_right(old_value, new_value):
    if old_value.isdigit() and new_value.isdigit():
        return int(new_value) * 2 == int(old_value) * 3
    return False  # 不是整数的派生字段值：按比值不对记，报告里点名


def classify_pair(old_fields, new_fields, tags):
    """返回 ('same',) 或 ('jia'|'yi'|'bing', 不等的键列表)。"""
    if pairing_key(old_fields, tags) != pairing_key(new_fields, tags):
        return ("jia", ["配对键对不上"])
    if [key for key, _ in old_fields] != [key for key, _ in new_fields]:
        return ("jia", ["字段名序列不同"])
    differing = [key for (key, old_value), (_, new_value) in zip(old_fields, new_fields) if old_value != new_value]
    if not differing:
        return ("same", [])
    derived = derived_field_names(new_fields)
    if any(key not in derived for key in differing):
        return ("jia", differing)
    old_map, new_map = dict(old_fields), dict(new_fields)
    if all(ratio_is_right(old_map[key], new_map[key]) for key in differing):
        return ("yi", differing)
    return ("bing", differing)


def pair_up(old_lines, new_lines):
    paired = list(zip(old_lines, new_lines))
    return paired, old_lines[len(paired):], new_lines[len(paired):]


def report_q11_q12(old_lines, new_lines, out):
    old_conclusions = [line for line in old_lines if line_name(parse_fields(line)) in CONCLUSION_NAMES]
    new_conclusions = [line for line in new_lines if line_name(parse_fields(line)) in CONCLUSION_NAMES]
    paired, old_extra, new_extra = pair_up(old_conclusions, new_conclusions)
    differing = []
    for index, (old_line, new_line) in enumerate(paired, start=1):
        kind, keys = classify_pair(parse_fields(old_line), parse_fields(new_line), CONCLUSION_TAGS)
        if kind != "same":
            differing.append((index, kind, keys, old_line, new_line))
    unpaired = len(old_extra) + len(new_extra)
    out.append("Q-11 conclusion_lines old=%d new=%d paired=%d differing_pairs=%d unpaired=%d differences=%d"
               % (len(old_conclusions), len(new_conclusions), len(paired), len(differing), unpaired, len(differing) + unpaired))
    for index, kind, keys, old_line, new_line in differing:
        out.append("Q-11 DIFF pair=%d class=%s differing_keys=%s" % (index, kind, ",".join(keys)))
        out.append("Q-11   OLD " + old_line)
        out.append("Q-11   NEW " + new_line)
    for line in old_extra:
        out.append("Q-11 UNPAIRED_OLD " + line)
    for line in new_extra:
        out.append("Q-11 UNPAIRED_NEW " + line)
    for scope, wanted in (("verdict(D=2 算出的格)", ("verdict", "discrimination_selfcheck")), ("geometry_sensitivity", ("geometry_sensitivity",))):
        counts = {"jia": 0, "yi": 0, "bing": 0}
        for _, kind, _, old_line, _ in differing:
            if line_name(parse_fields(old_line)) in wanted:
                counts[kind] += 1
        extra_in_scope = sum(1 for line in old_extra + new_extra if line_name(parse_fields(line)) in wanted)
        counts["jia"] += extra_in_scope  # 没配上的行归甲
        out.append("Q-12 scope=%s jia=%d yi=%d bing=%d" % (scope, counts["jia"], counts["yi"], counts["bing"]))
    return len(differing) + unpaired


def anchor_row_is_a_section_seven_row(old_fields, new_fields):
    anchor_id = (tag_value(new_fields, "id") or "").strip('"')
    wanted = SECTION_SEVEN_ANCHORS.get(anchor_id)
    if wanted is None:
        return False
    observed = (tag_value(new_fields, "observed") or "").strip('"')
    expected = (tag_value(new_fields, "expected") or "").strip('"')
    return observed == wanted and expected == wanted and tag_value(new_fields, "ok") == "true"


def report_q13(old_lines, new_lines, out):
    paired, old_extra, new_extra = pair_up(old_lines, new_lines)
    classes = {"same": 0, "yi_derived_ratio_ok": 0, "anchor_section_seven": 0, "other": 0}
    classes_devices_one = dict(classes)
    other_rows = []
    for index, (old_line, new_line) in enumerate(paired, start=1):
        old_fields, new_fields = parse_fields(old_line), parse_fields(new_line)
        kind, keys = classify_pair(old_fields, new_fields, FULL_LINE_TAGS)
        if kind == "same":
            label = "same"
        elif kind == "yi":
            label = "yi_derived_ratio_ok"
        elif line_name(new_fields) == "anchor" and kind == "jia" and anchor_row_is_a_section_seven_row(old_fields, new_fields):
            label = "anchor_section_seven"
        else:
            label = "other"
            other_rows.append((index, kind, keys, old_line, new_line))
        classes[label] += 1
        if tag_value(new_fields, "devices") == "1" and line_name(new_fields) in ("cell", "trajectory", "rounds"):
            classes_devices_one[label] += 1
    classes["other"] += len(old_extra) + len(new_extra)
    out.append("Q-13 lines old=%d new=%d paired=%d unpaired=%d" % (len(old_lines), len(new_lines), len(paired), len(old_extra) + len(new_extra)))
    out.append("Q-13 classes " + " ".join("%s=%d" % pair for pair in classes.items()))
    out.append("Q-13 devices=1 (cell,trajectory,rounds) classes " + " ".join("%s=%d" % pair for pair in classes_devices_one.items()))
    for index, kind, keys, old_line, new_line in other_rows:
        out.append("Q-13 OTHER pair=%d class=%s differing_keys=%s" % (index, kind, ",".join(keys)))
        out.append("Q-13   OLD " + old_line)
        out.append("Q-13   NEW " + new_line)
    for line in old_extra:
        out.append("Q-13 UNPAIRED_OLD " + line)
    for line in new_extra:
        out.append("Q-13 UNPAIRED_NEW " + line)


def main(old_path, new_path):
    out = []
    old_lines, new_lines = read_result_lines(old_path), read_result_lines(new_path)
    out.append("compare old=%s new=%s" % (old_path, new_path))
    report_q11_q12(old_lines, new_lines, out)
    report_q13(old_lines, new_lines, out)
    print("\n".join(out))


if __name__ == "__main__":
    main(sys.argv[1], sys.argv[2])
