// E169 停机条款 T1 的另一边：把登记第七节 A2 的十四条序列原样喂给 crates/singlefs-replay-prefix 今天的 replay_core，
// 打出它施加的记录数与发布数（发布数 = 施加的那几条里末条标志的个数；施加的是 above 里按次序的前 prefix_applied 条）。
use singlefs_replay_prefix::{replay_core, CoreRecord};

fn main() {
    // (序列号, 所选根 txg, [(jsn, txg, 事务号, 提交标记, 末条标志, 序号)], 点名单元验不过的 jsn)
    let sequences: Vec<(u32, u64, Vec<(u64, u64, u64, bool, bool, u32)>, Option<u64>)> = vec![
        (1, 0, vec![(1, 1, 1, true, true, 1)], None),
        (2, 0, vec![(1, 1, 1, false, false, 1), (2, 1, 1, true, true, 2)], None),
        (3, 0, vec![(1, 1, 1, false, false, 1)], None),
        (4, 0, vec![(1, 1, 1, false, false, 1), (2, 1, 2, true, false, 2), (3, 1, 1, true, true, 3)], None),
        (5, 0, vec![(1, 1, 1, true, false, 1), (2, 1, 2, true, true, 2)], None),
        (6, 0, vec![(1, 1, 1, true, false, 1), (2, 2, 2, true, true, 2)], None),
        (7, 0, vec![(1, 1, 1, true, true, 1), (2, 2, 2, true, true, 1)], None),
        (8, 0, vec![(1, 1, 1, true, true, 1), (3, 2, 2, true, true, 1)], None),
        (9, 0, vec![(1, 1, 1, true, true, 1), (2, 1, 1, true, false, 2)], None),
        (10, 0, vec![(1, 1, 1, true, true, 1)], Some(1)),
        (11, 0, vec![(1, 1, 1, false, true, 1)], None),
        (12, 0, vec![(1, 2, 1, true, true, 1)], None),
        (13, 1, vec![(1, 1, 1, true, true, 1), (2, 2, 2, true, true, 2)], None),
        (14, 0, vec![(1, 1, 1, true, false, 1), (2, 1, 2, true, true, 3)], None),
    ];
    for (number, root_txg, raw, unverified) in sequences {
        let records: Vec<CoreRecord<u64>> = raw
            .iter()
            .map(|(jsn, txg, transaction, is_commit, ends, ordinal)| CoreRecord {
                instance: 1,
                counter: *jsn,
                checkpoint_txg: *txg,
                transaction: *transaction,
                is_commit: *is_commit,
                ordinal_within_publish: *ordinal,
                ends_its_publish: *ends,
                new_root_segment: *jsn,
            })
            .collect();
        let jsns: Vec<u64> = raw.iter().map(|record| record.0).collect();
        let outcome = replay_core(1, root_txg, 0u64, &records, 65_536, true, |_index: usize| false, |index: usize| Some(jsns[index]) != unverified);
        match outcome {
            Ok(outcome) => {
                let above: Vec<&(u64, u64, u64, bool, bool, u32)> = raw.iter().filter(|record| record.1 > root_txg).collect();
                let publishes = above.iter().take(outcome.report.prefix_applied).filter(|record| record.4).count();
                println!("sequence={number} crates_records_applied={} crates_publishes_applied={publishes} rebuilt_segment={}", outcome.report.prefix_applied, outcome.rebuilt);
            }
            Err(_) => println!("sequence={number} crates_failure=yes"),
        }
    }
}
