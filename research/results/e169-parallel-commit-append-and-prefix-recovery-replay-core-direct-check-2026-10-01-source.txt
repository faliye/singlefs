// E169（并行提交的追加纪律与前缀恢复）路径 3 对路径 4 的直接对拍：读装置 `replay-core-inputs` 模式打出的每一份恢复输入
// （第一、二段每个崩溃状态的全部持久记录、严格连续段两种，按内容去重），原样喂给 crates/singlefs-replay-prefix 今天的 replay_core，
// 与同一行里模型丙0 的结果逐份比：报不报错、施加了几条、最后施加的那条 jsn、施加的最大事务号。
use singlefs_replay_prefix::{
    replay_core, CheckpointTxg, CoreRecord, InstanceGeneration, JournalCounter, JournalRecordOrdinalWithinPublish, JournalRecordPlaceInPublish, NamedUnitVerification,
    TransactionNumber,
};
use std::collections::{BTreeMap, HashMap};
use std::io::BufRead;

/// D23（journal 的角色与格式） 已定项 18 的在飞上限，与装置里 `IN_FLIGHT_RECORD_LIMIT` 同值（装置的 A6 断言那一判在全部状态上触发 0 次）。
const IN_FLIGHT_RECORD_LIMIT: usize = 65_536;
/// 模型是单实例（登记第十一节未建模）：所选根与每条记录都是实例 1。
const SINGLE_INSTANCE: u32 = 1;

#[derive(Debug, PartialEq, Eq)]
struct RecoveryResult {
    failure: bool,
    applied_records: usize,
    last_applied_jsn: Option<u64>,
    maximum_applied_transaction: u64,
}

#[derive(Default)]
struct Tally {
    inputs: u64,
    states: u64,
    mismatched_inputs: u64,
    mismatched_states: u64,
}

fn flag_of(text: &str) -> bool {
    match text {
        "1" => true,
        "0" => false,
        other => panic!("标志只有 0、1，这里是 {other}"),
    }
}

fn main() {
    let path = std::env::args().nth(1).expect("参数：装置 replay-core-inputs 模式的输出文件");
    let file = std::fs::File::open(&path).expect("输入文件读得开");
    let mut tallies: BTreeMap<String, Tally> = BTreeMap::new();
    let mut done_line_seen = false;
    for line in std::io::BufReader::new(file).lines() {
        let line = line.expect("输入文件逐行读得出");
        if line.starts_with("E7RESULT name=done ") {
            done_line_seen = true;
        }
        if !line.starts_with("E7RESULT name=replay_core_input ") {
            continue;
        }
        let fields: HashMap<&str, &str> = line.split(' ').skip(1).filter_map(|field| field.split_once('=')).collect();
        let input_kind = fields["input"].to_string();
        let root_txg: u64 = fields["root_txg"].parse().expect("root_txg 是整数");
        let states: u64 = fields["states"].parse().expect("states 是整数");
        let mut records: Vec<CoreRecord<u64>> = Vec::new();
        let mut named_units_verified: Vec<bool> = Vec::new();
        if fields["records"] != "none" {
            for record_text in fields["records"].split(',') {
                let parts: Vec<&str> = record_text.split(':').collect();
                assert_eq!(parts.len(), 7, "每条记录七段：jsn、txg、事务号、提交标记、末条标志、序号、点名单元验得过");
                let jsn: u64 = parts[0].parse().expect("jsn 是整数");
                records.push(CoreRecord {
                    instance: InstanceGeneration(SINGLE_INSTANCE),
                    counter: JournalCounter(jsn),
                    checkpoint_txg: CheckpointTxg(parts[1].parse().expect("txg 是整数")),
                    transaction: TransactionNumber(parts[2].parse().expect("事务号是整数")),
                    is_commit: flag_of(parts[3]),
                    ordinal_within_publish: JournalRecordOrdinalWithinPublish(parts[5].parse().expect("序号是整数")),
                    place_in_publish: if flag_of(parts[4]) { JournalRecordPlaceInPublish::LastRecordOfThePublish } else { JournalRecordPlaceInPublish::MoreRecordsOfThePublishFollow },
                    // 施加之后交回末条的这一项：取它的 jsn，比「最后施加的那条」。
                    new_root_source: jsn,
                });
                named_units_verified.push(flag_of(parts[6]));
            }
        }
        // replay_core 的前置条件（Verus 的 requires，运行时不查）：按 (实例代号, 计数器) 严格升序、计数器不是 u64 的最大值。
        assert!(records.windows(2).all(|pair| pair[0].counter.0 < pair[1].counter.0), "输入按 jsn 严格升序：{line}");
        assert!(records.iter().all(|record| record.counter.0 < u64::MAX));
        let outcome = replay_core(
            InstanceGeneration(SINGLE_INSTANCE),
            CheckpointTxg(root_txg),
            0u64,
            &records,
            IN_FLIGHT_RECORD_LIMIT,
            NamedUnitVerification::Verify,
            |_record_index: usize| false,
            |record_index: usize| named_units_verified[record_index],
        );
        let crates_result = match outcome {
            Ok(outcome) => RecoveryResult {
                failure: false,
                applied_records: outcome.report.prefix_applied,
                last_applied_jsn: if outcome.report.prefix_applied == 0 { None } else { Some(outcome.root_source_after_replay) },
                maximum_applied_transaction: outcome.report.maximum_applied_transaction.0,
            },
            Err(_) => RecoveryResult { failure: true, applied_records: 0, last_applied_jsn: None, maximum_applied_transaction: 0 },
        };
        let model_result = RecoveryResult {
            failure: fields["bing0_failure"] == "yes",
            applied_records: fields["bing0_applied_records"].parse().expect("施加条数是整数"),
            last_applied_jsn: match fields["bing0_last_applied_jsn"] {
                "none" => None,
                jsn => Some(jsn.parse().expect("jsn 是整数")),
            },
            maximum_applied_transaction: fields["bing0_maximum_applied_transaction"].parse().expect("事务号是整数"),
        };
        let tally = tallies.entry(input_kind.clone()).or_default();
        tally.inputs += 1;
        tally.states += states;
        if crates_result != model_result {
            tally.mismatched_inputs += 1;
            tally.mismatched_states += states;
            println!(
                "E169DIRECT name=mismatch input={input_kind} root_txg={root_txg} states={states} records={} crates={crates_result:?} model={model_result:?}",
                fields["records"]
            );
        }
    }
    assert!(done_line_seen, "输入文件有收尾的 done 行（没被截断）");
    for (input_kind, tally) in &tallies {
        println!(
            "E169DIRECT name=summary input={input_kind} distinct_inputs={} states={} mismatched_inputs={} mismatched_states={}",
            tally.inputs, tally.states, tally.mismatched_inputs, tally.mismatched_states
        );
    }
}
