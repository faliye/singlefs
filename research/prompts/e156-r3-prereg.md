#!/usr/bin/env python3
# E156 第 3 次重跑登记的独立锚点。常量从 D8 已定项 14、D3 已定项 7 / 10 / 11 手抄：叶宽 812、扇出 169、开放段起点 50240、
# 聚簇段 64 槽、码 3 容器与跨度 2 的单元起点 32768 对齐；「挂载后新开一段」取单元区内最低的 64 槽对齐全空段。
# 不 import 任何装置、不读 crates/。只模拟到登记要钉绝对值的那几步（第一个事务、H0 前 4 次、HK 的 O 与 E、S1(c) 场景），
# 那几步里开放段都没用完，所以用不上条款没写到的「开放段装不下时怎么办」。
W, FANOUT, UNIT_AREA_START, SEG = 812, 169, 50176, 64
DEV_SLOTS = (4 << 30) // 16384
def l1(s): return s // (W * FANOUT)
def path_nodes(leaves, devices=2):
    return devices * len(leaves) + devices * len({k * W // (W * FANOUT) for k in leaves}) + 1
print("R-3c pc_closed_form one_leaf =", path_nodes({61}), " two_leaves =", path_nodes({61, 62}),
      " leaves_in_two_internal =", path_nodes({61, 170}), " one_device =", path_nodes({61}, devices=1),
      " check_170_in_second_internal =", 170 * W >= W * FANOUT)
ALLOC = lambda: [("alloc", 1)] * 5          # 只在全部改动落在一片叶（两盘）时成立；下面每一步都断言这一点
class Pool:
    def __init__(self):
        self.occupied = {50176: 2, 50178: 1}   # mkfs：实例表第 0 片、第 0 版树表（后者在第一个事务被换下、已释放未回收）
        self.cursor = 50240                    # 开放段 [50240, 50304)
        self.segment_end = 50304
    def slot_free(self, s, span): return all(not any(o <= x < o + sp for o, sp in self.occupied.items()) for x in range(s, s + span))
    def data_slot(self):
        s = UNIT_AREA_START
        while True:
            in_open = self.segment_end - SEG <= s < self.segment_end
            if s % 2 == 0 and not in_open and self.slot_free(s, 2): return s
            s += 1
    def bump(self, span):
        if span == 2 and self.cursor % 2: self.cursor += 1
        assert self.cursor + span <= self.segment_end, "开放段装不下——超出本脚本能钉的范围"
        s = self.cursor; self.cursor += span; return s
    def new_segment(self):
        s = UNIT_AREA_START
        while any(not self.slot_free(x, 1) for x in range(s, s + SEG)): s += SEG
        self.cursor, self.segment_end = s, s + SEG
def publish(pool, roles, data=False):
    placed = []
    if data:
        s = pool.data_slot(); placed.append(("data", s, 2))
    for name, span in roles:
        placed.append((name, pool.bump(span), span))
    for _, s, sp in placed: pool.occupied[s] = sp
    return placed
O_ROLES = [("extent", 1), ("inode_leaf", 2), ("inode_root", 1)] + ALLOC() + [("accounting", 1), ("mapping", 1), ("tree_table", 1)]
E_ROLES = ALLOC() + [("accounting", 1), ("mapping", 1), ("tree_table", 1)]
W_ROLES = [("instance_table", 2)] + E_ROLES
def slots(units): return sum(sp for _, _, sp in units)
def one_leaf(units_list): return all(s // W == 61 and (s + sp - 1) // W == 61 for units in units_list for _, s, sp in units)
pool = Pool()
first = publish(pool, O_ROLES, data=True)
records_per_device = 2 + len(first)
item1 = 2 + 1 + slots(first)
print("K1-1 first_txn =", [(n, s, sp) for n, s, sp in first], " item1_slots =", item1, " item5_slots = 1",
      " item2_slots =", (DEV_SLOTS - UNIT_AREA_START) - item1, " one_leaf =", one_leaf([first]))
print("K1-2 records_per_device =", records_per_device, " both_devices =", 2 * records_per_device)
print("K1-3 referenced_at_beta0 =", 2 + slots(first), " item1_minus_item5 =", item1 - 1)
prev, h0 = first, []
for step in range(1, 5):
    cur = publish(pool, O_ROLES, data=True)
    assert one_leaf([prev, cur])
    released, delta = slots(prev), 2 * len(cur)
    h0.append((step, released, delta, [s for _, s, _ in cur][:2], max(s + sp - 1 for _, s, sp in cur)))
    prev = cur
print("R-4 h0_steps_1_to_4 (step, released_d0, record_delta, first_two_slots, last_slot) =", h0)
hk = Pool(); hk_first = publish(hk, O_ROLES, data=True); hk_o = publish(hk, O_ROLES, data=True); hk_e = publish(hk, E_ROLES)
assert one_leaf([hk_first, hk_o, hk_e])
released_e = sum(sp for n, s, sp in hk_o if n in ("alloc", "accounting", "mapping", "tree_table"))
print("R-5 hk_empty_publish_txg5 released_d0 =", released_e, " record_delta =", 2 * len(hk_e), " fixed_point_slots_one_leaf =", slots(hk_e))
c = Pool(); a = publish(c, O_ROLES, data=True); b = publish(c, O_ROLES, data=True)
c.new_segment(); w = publish(c, W_ROLES); e1 = publish(c, E_ROLES); e2 = publish(c, E_ROLES); cc = publish(c, O_ROLES, data=True)
assert one_leaf([a, b, w, e1, e2, cc])
print("R-6 s1c segment_after_remount =", (w[0][1], w[0][1] + SEG), " B =", slots(b), " W =", slots(w), " E1 =", slots(e1),
      " E2 =", slots(e2), " C =", slots(cc), " isolated_per_device =", slots(b) + slots(w) + slots(e1) + slots(e2) + slots(cc),
      " C_last_slot =", max(s + sp - 1 for _, s, sp in cc))
print("R-6 check_old_layout B=10 W=6 E=4 E=4 C=10 ->", 10 + 6 + 4 + 4 + 10)
print("A1 unit_area_slots =", DEV_SLOTS - UNIT_AREA_START, " A7 leaf_capacity =", (16384 - (86 + 2 * 10 + 29)) // 20)
