
    /// 探针 P1（正推腿加的）：豁免先于判重的次序。被抛弃根的落点 (s, 1) 被候选根 (mkfs 实例表 (起点, 跨 2)) 罩着，被抛弃根那一圈豁免它；
    /// 水位之上没施加的记录也点名 (s, 1)，C 那一圈只豁免现行账（空），所以 s 要被 C 那一圈隔离。
    #[test]
    fn probe_p1_a_slot_exempt_for_the_abandoned_roots_by_a_candidate_is_still_isolated_by_the_c_circle() {
        let pool = formatted_pool();
        let start = slot_after_the_unit_area_start(&pool, 0);
        let root = abandoned_root(&pool, 1, 8);
        let recorded = BTreeMap::from([(
            (root.instance, root.checkpoint_txg),
            vec![placement_on_device_zero(start, 1)],
        )]);
        let allocator = allocator_after_the_shadow_ledger(
            &pool,
            &ShadowLedgerInputs {
                roots: &[pool.make_filesystem_root, root],
                abandoned_instances: &[root.instance],
                current_records: &[],
                recorded_placements_of_abandoned_roots: &recorded,
                placements_recorded_for_abandoned_roots_whose_ring_slot_read_bad: &[],
                placements_named_by_unapplied_journal_records: &[placement_on_device_zero(start, 1)],
            },
        );
        assert_eq!(isolated_and_free_on_device_zero(&allocator, start), (true, false), "P1: C 那一圈隔离 s");
    }

    /// 探针 P3：落点罩着的一槽分配位置着（已释放、在 defer 窗口里：不在现行账的未释放记录里），表里记着的被抛弃根 A 的落点 (起点 + 1, 跨 2)
    /// 先过（只隔离分配位没置着的槽），读盘得来的被抛弃根 B（mkfs 那一版：实例表 (起点, 跨 2)、树表 (起点 + 2, 跨 1)）后过
    /// （整段隔离，不管分配位）。起点 + 2 被 A、B 同罩着：改前按起点槽判重时 B 的 (起点 + 2) 不在 A 的起点集合里、B 自己隔离；改后按槽判重时被 A 判重掉。
    #[test]
    fn probe_p3_a_slot_held_by_the_allocation_bit_and_referenced_by_two_abandoned_roots() {
        let pool = formatted_pool();
        let start = pool.unit_area_start.slot();
        let a = abandoned_root(&pool, 1, 8);
        let b = abandoned_root(&pool, 2, 9);
        let recorded = BTreeMap::from([(
            (a.instance, a.checkpoint_txg),
            vec![placement_on_device_zero(SlotNumber(start.0 + 1), 2)],
        )]);
        let mut allocator = PoolAllocator::new(
            pool.devices
                .iter()
                .map(|(identity, _)| {
                    DeviceFreeMap::with_unit_area_start(*identity, DEVICE_BYTES, pool.unit_area_start)
                })
                .collect(),
        );
        allocator.devices[0].mark_allocated(SlotNumber(start.0 + 2), 1);
        let is_abandoned = |root: &RootRecord| root.instance == a.instance || root.instance == b.instance;
        isolate_slots_referenced_only_by_abandoned_roots(
            &pool.devices,
            pool.unit_area_start,
            &mut allocator,
            &[a, b],
            &is_abandoned,
            CheckpointTxg(0),
            &[],
            &[],
            PlacementsOfAbandonedRoots::RecordedInTheRootRingTableOtherwiseReadFromTheirAccounts(&recorded),
            &mut BeforeTheOneReread::RereadImmediately,
            &[],
        )
        .expect("不拒");
        let outcome: Vec<bool> = (0..3)
            .map(|k| allocator.devices[0].is_isolated(SlotNumber(start.0 + k)))
            .collect();
        println!("P3 outcome (isolated on dev0 for start+0..2) = {outcome:?}");
        assert_eq!(outcome, vec![true, true, true], "P3: B 的整段隔离要罩到起点 + 2");
    }
