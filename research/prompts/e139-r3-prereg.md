def slot(t, S): return (t % 3) * S + (t // 3) % S
def disk(s, S): return 1 if s // S == 1 else 0
def build(S, publishes, floors, torn=()):
    ring = {s: (0, 0, True) for s in range(3 * S)}  # slot -> (txg, F carried, self_verified)
    for t in sorted(set(publishes) | set(torn)):
        ring[slot(t, S)] = (t, floors.get(t, 0), t not in torn)
    return ring
def valid(ring): return {s: v for s, v in ring.items() if v[2]}
def old_floors(ring, S):
    v = valid(ring)
    visible_f = [f for (_, f, ok) in ring.values() if ok]
    candidacy = max(visible_f) if visible_f else 0
    per_disk = {}
    for s, (t, f, _) in v.items():
        per_disk[disk(s, S)] = max(per_disk.get(disk(s, S), 0), f)
    settled = min(min(per_disk.get(d, 0) for d in (0, 1)), candidacy)
    return candidacy, settled
def core_floors(ring, S, sysconfig):
    v = valid(ring)
    newest = {}
    for s, (t, f, _) in v.items():
        d = disk(s, S)
        if d not in newest or t > newest[d][0]: newest[d] = (t, f)
    f_eff = max([f for (_, f) in newest.values()] + [sysconfig])
    m = min(newest[d][0] for d in (0, 1)) if len(newest) == 2 else None
    return f_eff, m

    for s, (_, _, ok) in ring.items():
        if ok: continue
        out[s] = next(t for t in range(min(3 * S, first_txg)) if slot(t, S) == s)
    return out
def oldest_valid(ring): return min(t for (t, _, ok) in ring.values() if ok)

    first_txg = max(publishes) + 1

    f_eff, m = core_floors(ring, S, sysconfig)

## 十四、并行与挂钟

整份登记的预计挂钟在 2 小时内（推的，各段的数与依据在下面五行里）。

独立单位：每一次模拟（臂 × 世界 × S × 丢盘 × 开销对 × 失败组合 × 崩溃窗口 × 种子）互相独立，只在行内按种子合计；单测、变异表每一条互相独立
线程：编译、单测与变异表经 `bash scripts/hosts/capped.sh 31` 起，31 取登记值 `.claude/heavy-thread-default`；产物二进制只有一条线程（装置里没有并行，`grep -n 'thread\|rayon\|par_iter'` 零命中），这一段挂钟在 2 小时内，不为它加并行
串行：改前复现（第六节第 1 行）必须在改装置之前跑完；产物一次运行内串行，逐行比的旧那一边取同一次运行里重出的 `f_` 行（第一节读法第 6 条），拆成几次运行就要另比；变异表在单测全绿之后跑
预计挂钟：第一段——改前复现约 10 分钟（第 2 次登记第十二节「整份产物要十分钟上下」）；新产物约 20 分钟（推的：第 2 次的 10 分钟加上两条 CG 被测臂进 RGall 一族的全部格、约与 RGall 一族等量，短臂进三类世界、两族保留池扫描各几十次近满盘运行）；单测 1 分钟内；变异表 53 条（原 41 行加第九节 12 条）约 60 分钟内（推的：每条一次编译加一次单测，31 条线程分两到三轮）。第二段约 10 分钟（一条臂进甲类格，推的）。合计不超过 2 小时
