# E142 第 23 次差分重跑登记第七节 7.2 的独立锚点：系统配置槽里 T_time 那 4 字节的偏移与值。
# 偏移由 layout/01 系统配置字段表的「宽」一列逐行累加现算（不取装置与 crates/ 的常量）；值 = 5 秒 × 1000 = 5000 毫秒（用户 2026-10-05 定案）。
# 不 import 装置、不读 crates/ 代码。用法（在仓根跑，第一参数缺省时读 .claude/kb/layout/01-first-txn.md）：
#   python3 ttime_anchor.py                                   # 只打锚点与自检
#   python3 ttime_anchor.py --decode <产物文件> [<行名>]        # 从产物里两侧（装置 region= / crates 无 region=）的系统配置槽 1 写解出 magic 与 T_time

#   python3 ttime_anchor.py --diff <新产物> <旧产物> [<行名>]   # 装置一侧同一（设备，偏移，长度）的系统配置槽 1 写逐字节比，列不同字节的偏移段，
#                                                             # 判它们是不是都落在 T_time 与整槽校验和 CRC 那两段里
# 弄坏开关 TTIME_ANCHOR_BREAK=1：累加时漏掉 T_time 前一行（固定结构槽距 4），自检必须打 False。
import os
import re
import sys

LAYOUT = ".claude/kb/layout/01-first-txn.md"
BREAK = os.environ.get("TTIME_ANCHOR_BREAK") == "1"
SLOT_SPACING = 4096          # layout/01 第 41 行：固定结构槽距 4096，槽 1 在设备内偏移 4096
SLOT_BYTES = 4096            # layout/01 第 40 行：槽宽 4096
T15_SLOT_INDEX = 1           # layout/01 第 81 行：t15 写系统配置槽 1、世代号 5


def field_widths(path):
    rows = []
    inside = False
    with open(path, encoding="utf-8") as handle:
        for line in handle:
            if line.startswith("**系统配置字段表**"):
                inside = True
                continue
            if inside and line.startswith("合计 **489 字节**"):
                break
            if not inside or not line.startswith("| ") or line.startswith("| 段 ") or line.startswith("|---"):
                continue
            cells = [cell.strip() for cell in line.strip().strip("|").split("|")]
            width_text = cells[2]
            match = re.fullmatch(r"(\d+)(?: × (\d+))?", width_text)
            if match is None:
                raise SystemExit(f"认不出宽：{line[:80]}")
            width = int(match.group(1)) * (int(match.group(2)) if match.group(2) else 1)
            rows.append((cells[1], width))
    return rows


def offsets(rows):
    found = {}
    position = 0
    for index, (name, width) in enumerate(rows):
        if BREAK and index + 1 < len(rows) and rows[index + 1][0].startswith("T_time"):
            continue
        found.setdefault(name, position)
        position += width
    return found, position


def anchors(path):
    rows = field_widths(path)
    found, total = offsets(rows)
    t_time = next(value for key, value in found.items() if key.startswith("T_time"))
    checksum = found["整槽校验和"]
    print(f"A7 rows {len(rows)} total {total} expected 489 {total == 489}")
    print(f"A7 t_time_offset {t_time} expected 433 {t_time == 433}")
    print(f"A7 slot_checksum_offset {checksum} crc_bytes [{checksum},{checksum + 4}) expected 155 {checksum == 155}")
    value = 5 * 1000
    print(f"A8 t_time_value {value} le_hex {value.to_bytes(4, 'little').hex()} old_value 5 old_le_hex {(5).to_bytes(4, 'little').hex()}")
    print(f"A9 t15_slot device_offset {T15_SLOT_INDEX * SLOT_SPACING} length {SLOT_BYTES} devices 0,1")
    return t_time, checksum


def slot_lines(path, line_name):
    with open(path, encoding="utf-8", errors="replace") as handle:
        for line in handle:
            if not line.startswith(f"E7RESULT name={line_name} "):
                continue
            fields = dict(token.split("=", 1) for token in line.split() if "=" in token)
            if fields.get("offset") != str(T15_SLOT_INDEX * SLOT_SPACING) or fields.get("length") != str(SLOT_BYTES):
                continue
            yield fields


def decode(path, line_name, t_time):
    for fields in slot_lines(path, line_name):
        side = "model" if "region" in fields else "crates"
        slot = bytes.fromhex(fields["hexadecimal"])
        magic = slot[0:4].decode("ascii", errors="replace")
        value = int.from_bytes(slot[t_time:t_time + 4], "little")
        print(f"TTIME side={side} device={fields['device']} magic={magic} t_time={value} le_hex={slot[t_time:t_time + 4].hex()}")


def diff(new_path, old_path, line_name, t_time, checksum):
    old = {(f["device"]): bytes.fromhex(f["hexadecimal"]) for f in slot_lines(old_path, line_name) if "region" in f}
    for fields in slot_lines(new_path, line_name):
        if "region" not in fields:
            continue
        new = bytes.fromhex(fields["hexadecimal"])
        before = old.get(fields["device"])
        if before is None:
            print(f"TTIME_DIFF device={fields['device']} unmatched_old=1")
            continue
        changed = [index for index in range(len(new)) if new[index] != before[index]]
        allowed = set(range(t_time, t_time + 4)) | set(range(checksum, checksum + 4))
        outside = [index for index in changed if index not in allowed]
        print(f"TTIME_DIFF device={fields['device']} changed_bytes={len(changed)} offsets={','.join(map(str, changed)) or 'none'} outside_t_time_and_crc={len(outside)}")


def main():
    t_time, checksum = anchors(LAYOUT)
    if len(sys.argv) in (3, 4) and sys.argv[1] == "--decode":
        decode(sys.argv[2], sys.argv[3] if len(sys.argv) == 4 else "device_region_bytes", t_time)
    elif len(sys.argv) in (4, 5) and sys.argv[1] == "--diff":
        diff(sys.argv[2], sys.argv[3], sys.argv[4] if len(sys.argv) == 5 else "device_region_bytes", t_time, checksum)
