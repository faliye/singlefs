import math

LEAF_ALLOC = (16384 - (115 + 2 * 10)) // 20
SEG = 64

def cont(n, cov, groups, gaps):
    if n == 0 or not groups:
        return 0.0
    t = 1.0 + sum((k - 1.0) / cov for k in groups) + sum(min(1.0, gp / cov) for gp in gaps)
    return min(float(n), t)

def scat(k, region, cov):
    if k <= 0:
        return 0.0
    m = max(1.0, region / cov)
    return m * (1.0 - (1.0 - 1.0 / m) ** k)

def scat_cont(n, cov, G, s, region):
    if G <= 0 or s <= 0:
        return 0.0
    m = max(1.0, region / cov)
    c = min(m, 1.0 + (s - 1.0) / cov)
    return min(float(n), m * (1.0 - (1.0 - c / m) ** G))

    n = float(n)
    surv = 1.0 - min(1.0, c / n)
    for s in ss:
        surv *= 1.0 - min(1.0, s / n)
    return n * (1.0 - surv)

    """② 触到的块数 K = sum_{j>=2} n_p (1 - exp(-lambda_j / n_p))，lambda_j = X q (1-q)^(j-1)。

    if X <= 0:
        return 0.0
    n_p = 1.0 if seg is None else max(1.0, math.ceil(S / seg))
    if q <= 1e-12:
        return X * (1.0 - q)

            break

def epoch_value(q, lam_fn, S=400.0, seg=64, poisson=True):
    n_p = max(1.0, math.ceil(S / seg)); p = S / n_p
    K = 0.0
    for j in range(2, 200000):
        lam = lam_fn(q, j)
        if lam / n_p < 1e-12:
            break

    nf = a.scat_cont(n, cov, K, p, R)
