def sequence(n_unlinks, first_u=0, compact_by_unlink=False, p=64):
    """第 j 次 unlink（j = 0, 1, …）：意图在 u = first_u + j、a_T = 1；最后一批在 u = first_u + j + 1、a_T = 0。
