
/// 正推腿的差分探针用（只在草稿副本里加，不入库）：`chain_record_view` 的结局编成 0 最后一片、1 有下一片、2 解不开。
#[must_use]
pub fn chain_record_view_code_for_the_forward_leg_probe(record: &[u8]) -> u8 {
    match chain_record_view(record) {
        ChainRecordView::LastPage => 0,
        ChainRecordView::NextPage(_) => 1,
        ChainRecordView::Malformed => 2,
    }
}
