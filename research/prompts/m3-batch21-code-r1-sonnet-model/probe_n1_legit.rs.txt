//! 正推腿的探针（N1）：合法删除全程的每一版镜像上，I-9.5、I-4.7、I-4.6、I-3.4 的判定与理由。
//! 看新加的 extent 树反方向配对有没有在合法状态上把 I-4.7、I-4.6 变成「判不了」（不适用带理由）。
#![allow(unused)]
mod common;
mod common_deletion_in_progress;
mod common_pointer_head_corruption;

use common_deletion_in_progress::{
    pool_with_a_three_unit_file, publish_new_inodes_and_advance_the_pool, publish_the_intent,
    publish_the_next_step_in_batches_of_one, verdict_of, violated_invariants,
};
use singlefs_harness::memory_pool::MemoryPool;

fn report(stage: &str, image: &MemoryPool) {
    let line: Vec<String> = ["I-9.5", "I-4.7", "I-4.6", "I-3.4"]
        .iter()
        .map(|invariant| format!("{invariant}={:?}", verdict_of(image, invariant)))
        .collect();
    println!(
        "PROBE-N1-LEGIT stage={stage} violated={:?} {}",
        violated_invariants(image),
        line.join(" ")
    );
}

#[test]
fn probe_n1_legit_states_of_a_deletion_from_the_entry_to_the_new_inode() {
    let mut pool = pool_with_a_three_unit_file("probe-n1-legit");
    report("three-unit-file", &pool.memory_pool());
    publish_the_intent(&mut pool);
    report("after-intent", &pool.memory_pool());
    for batch in 1..=3 {
        let last_batch = publish_the_next_step_in_batches_of_one(&mut pool);
        println!(
            "PROBE-N1-LEGIT-SHAPE batch={batch} inode_record_is_none={} deleted_inodes_is_empty={} data_pointers={} lower_nodes={} upper_nodes={}",
            last_batch.inode_record.is_none(),
            last_batch.deleted_inodes.is_empty(),
            last_batch.data_pointers.len(),
            last_batch.extent_tree.lower_nodes.len(),
            last_batch.extent_tree.upper_nodes.len()
        );
        report(&format!("after-batch-{batch}"), &pool.memory_pool());
    }
    publish_new_inodes_and_advance_the_pool(&mut pool, 1);
    report("after-new-inode-on-the-emptied-version", &pool.memory_pool());
}
