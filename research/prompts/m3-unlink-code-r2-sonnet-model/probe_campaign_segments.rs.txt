//! 探针（草稿副本，不入仓）：在 harness 档复核 checker 档战役用例写死的段形与写数。
use singlefs_core::address::{CheckpointTxg, InstanceGeneration};
use singlefs_core::admission::SpaceAdmission;
use singlefs_harness::history::{
    execute_history_with, ContentChoice, ContentLength, GeneratedHistory, HistoryDeviceWidth,
    HistoryEnding, HistoryExecution, HistoryOperation, HistorySeed, HistoryStartingPoint,
    PerStepChecker,
};
use singlefs_harness::memory_pool::{root_identity_written_by, writes_and_segments, RetainedWrite};
use singlefs_harness::segments::{segment_sizes_text, split_into_segments, StepKind};
use singlefs_harness::SharedStream;

const EXECUTION: HistoryExecution = HistoryExecution {
    per_step_checker: PerStepChecker::Skipped,
    device_width: HistoryDeviceWidth::FourGibibytes,
    space_admission: SpaceAdmission::JudgedByTheFormula,
};

fn overwrite_choosing(selector: u64) -> HistoryOperation {
    HistoryOperation::PublishOverwrite(ContentChoice {
        length: ContentLength::InsideOneDataUnit { selector },
        fill_seed: selector,
    })
}

fn run(history: &GeneratedHistory) -> (Vec<RetainedWrite>, Vec<Vec<usize>>) {
    let stream = SharedStream::retaining_contents();
    let run = execute_history_with(history, EXECUTION, &stream, &mut |_| {});
    assert_eq!(run.ending, HistoryEnding::Completed);
    writes_and_segments(&stream.retained_operations(), &EXECUTION.device_width.fixed_geometry())
}

fn root_of(w: &RetainedWrite) -> Option<(u32, u64)> {
    if w.kind != StepKind::RootRecordFua {
        return None;
    }
    w.bytes().map(|b| {
        let (i, t) = root_identity_written_by(b);
        (i.0, t.0)
    })
}

#[test]
fn probe_campaign_segment_claims() {
    // 历史 A：起点发完第一个文件，再覆盖写 5、7。
    let a = GeneratedHistory {
        seed: HistorySeed(0),
        starting_point: HistoryStartingPoint::AfterFirstFile,
        operations: vec![overwrite_choosing(5), overwrite_choosing(7)],
    };
    let (writes, segments) = run(&a);
    println!("PROBE A segments={} sizes={}", segments.len(), segment_sizes_text(&segments.iter().map(|s| s.iter().map(|i| writes[*i].kind).collect()).collect::<Vec<Vec<StepKind>>>()));
    let f21 = segments[21][0];
    println!("PROBE A seg21 first3 kinds={:?} root_of_third={:?}", writes[f21..f21 + 3].iter().map(|w| w.kind).collect::<Vec<_>>(), root_of(&writes[f21 + 2]));
    let f12 = segments[12][0];
    let roots: Vec<(usize, (u32, u64))> = (f12..f12 + 56).filter_map(|i| root_of(&writes[i]).map(|r| (i - f12, r))).collect();
    println!("PROBE A seg12 first2 kinds={:?} same_offset={} roots_in_56={:?} root_of_56th={:?}", writes[f12..f12 + 2].iter().map(|w| (w.kind, w.device.0, w.offset.0)).collect::<Vec<_>>(), writes[f12].offset == writes[f12 + 1].offset, roots, root_of(&writes[f12 + 55]));
    // 历史 B：24 次覆盖写。
    let b = GeneratedHistory {
        seed: HistorySeed(0),
        starting_point: HistoryStartingPoint::AfterFirstFile,
        operations: (0..24).map(|i| overwrite_choosing(5 + 2 * i)).collect(),
    };
    let (writes_b, segments_b) = run(&b);
    let f12b = segments_b[12][0];
    println!("PROBE B writes_total={} from_seg12={} writes[f12b+4..6]={:?}", writes_b.len(), writes_b.len() - f12b, writes_b[f12b + 4..f12b + 6].iter().map(|w| (w.kind, w.device.0, w.offset.0)).collect::<Vec<_>>());
    let first_file_root = writes_b.iter().position(|w| root_of(w) == Some((1, 3))).expect("(1,3) 根槽写");
    println!("PROBE B first_file_root_idx={} f12b+5<root={} overwritten_later={}", first_file_root, f12b + 5 < first_file_root, writes_b[first_file_root + 1..].iter().any(|w| w.kind == StepKind::RootRecordFua && w.device == writes_b[first_file_root].device && w.offset == writes_b[first_file_root].offset));
    let later_hits = writes_b[f12b + 6..].iter().filter(|l| (l.device == writes_b[f12b + 4].device && l.offset == writes_b[f12b + 4].offset) || (l.device == writes_b[f12b + 5].device && l.offset == writes_b[f12b + 5].offset)).count();
    println!("PROBE B later_writes_on_withheld_locations={later_hits}");
    let _ = (InstanceGeneration(0), CheckpointTxg(0), split_into_segments);
    // 历史 C：零单元发布与第一个文件的段序列（战役用例 crash_states_land_in_zero_unit_publishes...）。
    let c = GeneratedHistory {
        seed: HistorySeed(0),
        starting_point: HistoryStartingPoint::AfterMakeFilesystem,
        operations: vec![
            HistoryOperation::CloseAndMountWritable,
            HistoryOperation::PublishWithoutUnits,
            HistoryOperation::ColdStartRecover,
            HistoryOperation::CloseAndMountWritable,
            HistoryOperation::PublishFirstFile(ContentChoice { length: ContentLength::InsideOneDataUnit { selector: 3 }, fill_seed: 1 }),
            HistoryOperation::CrashRecoveryAbandoningTheNewestRoot,
            HistoryOperation::CloseAndMountWritable,
        ],
    };
    let stream = SharedStream::retaining_contents();
    let run_c = execute_history_with(&c, EXECUTION, &stream, &mut |_| {});
    let (writes_c, segments_c) = writes_and_segments(&stream.retained_operations(), &EXECUTION.device_width.fixed_geometry());
    println!("PROBE C ending={:?} outcomes={:?}", run_c.ending, run_c.outcomes);
    println!("PROBE C segment sizes={:?}", segments_c.iter().map(Vec::len).collect::<Vec<_>>());
    let _ = writes_c;
}
