// 追加在冻结副本 crates/singlefs-harness/tests/admission_raises_the_floor_before_refusing.rs 末尾的探针（草稿副本里跑过，不入仓）
// 一、各次发布按结构种类的写数；二、续做里推过抬 F 之后设备写错：错里写账与设备写数的差、次序按 txg；三、挂载透明 F 判在续做推过抬 F 的挂载上
fn probe_print_kinds(label: &str, w: &singlefs_core::write_accounting::WritesByStructureKind) {
    let mut s = String::new();
    for k in singlefs_core::write_accounting::WrittenStructureKind::IN_REPORT_ORDER {
        let n = w.of(k).write_calls;
        if n > 0 {
            s += &format!("{k:?}={n} ");
        }
    }
    println!("KINDS {label}: {s}");
}

#[test]
fn probe_kinds_of_each_publish_of_a_mount_with_a_continuation() {
    let (mut pool, _) = pool_crashed_between_deletion_batches_on_narrow_devices(0);
    let mounted = mount_after_the_crash_with_batches_of_one(&mut pool).expect("mount");
    let o = &mounted.output;
    probe_print_kinds("row", o.row_publish.writes());
    for (i, w) in o.warm_up_publishes.iter().enumerate() {
        probe_print_kinds(&format!("warm{i}"), w.writes());
    }
    for (i, r) in o.space_admission.floor_raises().iter().enumerate() {
        probe_print_kinds(&format!("admraise{i}-sysconf"), &r.system_configuration_writes_before_the_first_publish);
        for (j, p) in r.publishes.iter().enumerate() {
            probe_print_kinds(&format!("admraise{i}-pub{j}"), &p.writes);
        }
    }
    for (i, s) in o.deletion_queue_continuation.iter().enumerate() {
        println!("STEP {i} txg={}", s.root.checkpoint_txg.0);
        probe_print_kinds(&format!("step{i}"), &s.writes);
    }
    for (i, r) in o.floor_raises_pushed_by_the_continuation.iter().enumerate() {
        probe_print_kinds(&format!("contraise{i}-sysconf"), &r.system_configuration_writes_before_the_first_publish);
        for (j, p) in r.publishes.iter().enumerate() {
            println!("RAISEPUB {i}.{j} txg={}", p.root.checkpoint_txg.0);
            probe_print_kinds(&format!("contraise{i}-pub{j}"), &p.writes);
        }
    }
}

struct ProbeFailingWrite<Inner> {
    inner: Inner,
    writes_so_far: std::rc::Rc<std::cell::Cell<u64>>,
    failing_write: u64,
}

impl<Inner: singlefs_core::block_device::BlockDevice> singlefs_core::block_device::BlockDevice
    for ProbeFailingWrite<Inner>
{
    fn read_at(
        &self,
        offset: singlefs_core::address::DeviceOffsetInBytes,
        buffer: &mut [u8],
    ) -> Result<(), singlefs_core::block_device::BlockDeviceError> {
        self.inner.read_at(offset, buffer)
    }
    fn write_at(
        &mut self,
        offset: singlefs_core::address::DeviceOffsetInBytes,
        bytes: &[u8],
        durability: singlefs_core::block_device::WriteDurability,
    ) -> Result<(), singlefs_core::block_device::BlockDeviceError> {
        let this_write = self.writes_so_far.get() + 1;
        self.writes_so_far.set(this_write);
        if this_write == self.failing_write {
            return Err(singlefs_core::block_device::BlockDeviceError::InputOutput(
                std::io::Error::other("探针注入的写错"),
            ));
        }
        self.inner.write_at(offset, bytes, durability)
    }
    fn write_zeroes_at(
        &mut self,
        offset: singlefs_core::address::DeviceOffsetInBytes,
        length: u64,
    ) -> Result<(), singlefs_core::block_device::BlockDeviceError> {
        self.writes_so_far.set(self.writes_so_far.get() + 1);
        self.inner.write_zeroes_at(offset, length)
    }
    fn barrier(&mut self) -> Result<(), singlefs_core::block_device::BlockDeviceError> {
        self.inner.barrier()
    }
    fn probe_physical_block_size(&self) -> singlefs_core::block_device::PhysicalBlockSizeInBytes {
        self.inner.probe_physical_block_size()
    }
    fn size_in_bytes(&self) -> u64 {
        self.inner.size_in_bytes()
    }
}

#[test]
fn probe_account_after_a_floor_raise_pushed_inside_the_continuation_then_a_device_error() {
    let (pool, _) = pool_crashed_between_deletion_batches_on_narrow_devices(0);
    let image = pool.image();
    let count = |failing_write: u64| {
        let writes_so_far = std::rc::Rc::new(std::cell::Cell::new(0u64));
        let mut devices: Vec<(DeviceIdentity, ProbeFailingWrite<SparseBlockDevice>)> =
            plain_devices_on(&image)
                .into_iter()
                .map(|(identity, device)| {
                    (
                        identity,
                        ProbeFailingWrite {
                            inner: device,
                            writes_so_far: writes_so_far.clone(),
                            failing_write,
                        },
                    )
                })
                .collect();
        let result = mount_writable_with_test_only_switches_and_deletion_switches(
            &pool.parameters,
            &mut devices,
            SpaceAdmission::JudgedByTheFormula,
            ShadowLedger::On,
            deletion_switches_with_batches_of_one(),
        );
        (result, writes_so_far.get())
    };
    let (counted, total_writes) = count(u64::MAX);
    let mounted = counted.expect("数写的那一次挂载做成");
    let o = &mounted.output;
    let calls = |w: &singlefs_core::write_accounting::WritesByStructureKind| w.total().write_calls;
    let mut before: u64 = 2 + calls(o.row_publish.writes());
    for w in &o.warm_up_publishes {
        before += calls(w.writes());
    }
    for r in o.space_admission.floor_raises() {
        before += calls(&r.system_configuration_writes_before_the_first_publish);
        for p in &r.publishes {
            before += calls(&p.writes);
        }
    }
    let first_raise_txg = o.floor_raises_pushed_by_the_continuation[0].publishes[0].root.checkpoint_txg.0;
    let last_raise_txg = o.floor_raises_pushed_by_the_continuation[0].publishes.last().unwrap().root.checkpoint_txg.0;
    for s in &o.deletion_queue_continuation {
        if s.root.checkpoint_txg.0 < first_raise_txg {
            before += calls(&s.writes);
        }
    }
    let raise = &o.floor_raises_pushed_by_the_continuation[0];
    let sysconf = calls(&raise.system_configuration_writes_before_the_first_publish);
    before += sysconf;
    for p in &raise.publishes {
        before += calls(&p.writes);
    }
    println!("PROBE total_writes_of_the_whole_mount={total_writes} first_raise_txg={first_raise_txg} last_raise_txg={last_raise_txg} raise_sysconf_writes={sysconf}");
    let failing_write = before + 1;
    let (failed_run, issued) = count(failing_write);
    let failed = match failed_run {
        Err(MountError::Publish(failed)) => failed,
        Err(other) => panic!("探针：想要 Publish，得到 {other:?}"),
        Ok(_) => panic!("探针：挂载却做成了"),
    };
    let sum_persisted: u64 = failed.writes_of_persisted_publishes.iter().map(|w| w.total().write_calls).sum();
    println!(
        "PROBE failing_write={failing_write} device_writes_issued_before_failure={} account_persisted_sum={sum_persisted} acquisition=2 persisted_len={} before_first_publish={} gap={}",
        issued - 1,
        failed.writes_of_persisted_publishes.len(),
        failed.writes_before_the_first_publish.total().write_calls,
        (issued - 1) as i64 - (sum_persisted as i64 + 2)
    );
    // 第二次：失败落在续做推完那一串之后、又发成一步的下一步（步 12，txg 25）的第一个写上，核错里写账的次序是 txg 的先后。
    let step_after_the_raise = o
        .deletion_queue_continuation
        .iter()
        .find(|s| s.root.checkpoint_txg.0 > last_raise_txg)
        .expect("推完那一串之后还有续做的步");
    let second_failing_write = before + calls(&step_after_the_raise.writes) + 1;
    let (second_run, second_issued) = count(second_failing_write);
    let second_failed = match second_run {
        Err(MountError::Publish(failed)) => failed,
        other => panic!("探针二：想要 Publish，得到 {:?}", other.err()),
    };
    let mut expected: Vec<u64> = vec![calls(o.row_publish.writes())];
    for w in &o.warm_up_publishes {
        expected.push(calls(w.writes()));
    }
    for r in o.space_admission.floor_raises() {
        for p in &r.publishes {
            expected.push(calls(&p.writes));
        }
    }
    for s in &o.deletion_queue_continuation {
        if s.root.checkpoint_txg.0 < first_raise_txg {
            expected.push(calls(&s.writes));
        }
    }
    for p in &raise.publishes {
        expected.push(calls(&p.writes));
    }
    expected.push(calls(&step_after_the_raise.writes));
    let actual: Vec<u64> = second_failed
        .writes_of_persisted_publishes
        .iter()
        .map(|w| w.total().write_calls)
        .collect();
    println!("PROBE2 issued_before_failure={} expected_by_txg={expected:?} actual={actual:?} equal={}", second_issued - 1, expected == actual);
    assert_eq!(actual, expected, "错里写账按 txg 的先后");
}

#[test]
fn probe_mount_transparency_floor_when_only_the_continuation_pushed_a_floor_raise() {
    use singlefs_harness::mount_transparency::{
        judge_mount_transparency, read_recovery_fields, ComparedVersion, FloorRaisedByTheMount,
        MountTransparencyAction,
    };
    let (mut pool, _) = pool_crashed_between_deletion_batches_on_narrow_devices(0);
    let image_before = pool.image();
    let before = read_recovery_fields(&image_before, ComparedVersion::TheRecoveredVersion);
    let mounted = mount_after_the_crash_with_batches_of_one(&mut pool).expect("mount");
    let floor = FloorRaisedByTheMount::of_the_mount_output(&mounted.output);
    let admission_raises = mounted.output.space_admission.floor_raises().len();
    let continuation_raises = mounted.output.floor_raises_pushed_by_the_continuation.len();
    let image_after = pool.image();
    let after = read_recovery_fields(&image_after, ComparedVersion::TheRecoveredVersion);
    let verdict = judge_mount_transparency(&before, &after, MountTransparencyAction::MountAlone(floor));
    println!(
        "PROBE3 floor_relation_input={floor:?} admission_raises={admission_raises} continuation_raises={continuation_raises} effective_floor_before={:?} effective_floor_after={:?} violated={:#?}",
        before.effective_rollback_floor, after.effective_rollback_floor, verdict.violated_fields()
    );
}
