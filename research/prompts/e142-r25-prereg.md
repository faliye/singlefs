def compare(old_text, new_text, profile):
    out = []
    old_raw, new_raw = first_half(old_text), first_half(new_text)

def selftest():
    all_ok = True
    for label, old_variant, new_variant, profile, expected in SAMPLES:
        lines, passed, total = compare(synthetic(old_variant), synthetic(new_variant), profile)
        verdict = next(l for l in lines if l.startswith("R25_VERDICT "))

        print(f"PC_G25 sample={label} pass={str(passed).lower()} expected={str(expected).lower()} ok={ok} changed_lines_total={total} {verdict.split(' ', 3)[3]}")
