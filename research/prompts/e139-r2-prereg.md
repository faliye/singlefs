def slot(t, S): return (t % 3) * S + (t // 3) % S
def disk_of_slot(s, S): return 1 if s // S == 1 else 0
def ring_after(S, publishes, failed):
    ring = {s: 0 for s in range(3 * S)}
    for t in publishes:
        if t in failed: continue
        ring[slot(t, S)] = t
    return ring
