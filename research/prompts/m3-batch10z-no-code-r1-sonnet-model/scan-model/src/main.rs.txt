//! 正推腿的模型（m3-batch10z-no-code-r1）：把第十五块 checker 档两条扫源码用例逐字搬出来，喂任意一份源文件，报 GREEN 或 RED。
//! 搬运来源：冻结副本 crates/singlefs-checker-tier/src/gpu_unit_checks.rs 的 `the_card_pool_hands_its_kept_devices_to_the_ledger_before_booking`
//! 与 crash_amplification.rs 的 `whole_name_uses_of_the_stop_request`、`code_lines_outside_the_tests`、`every_layer_handed_the_stop_request_hands_it_on_or_asks_it`。
const TESTS_MODULE_OPENING: &str = "#[cfg(test)]\nmod tests \u{7b}";
const TOP_LEVEL_CLOSING_BRACE_LINE: &str = "\n\u{7d}\n";

fn gpu(source: &str) -> Result<(), String> {
    let code = source.split(TESTS_MODULE_OPENING).next().expect("split 至少交一段");
    let signature = "fn take_in_the_devices_found_idle(&self) \u{7b}";
    if code.matches(signature).count() != 1 {
        return Err(format!("RED: 源码里恰好一处 {signature}"));
    }
    let start = code.find(signature).ok_or(format!("源码里没有 {signature}"))?;
    let after = &code[start + signature.len()..];
    let body = &after[..after.find("\n            \u{7d}\n").ok_or("借之前那一步以缩进十二格的右括号收尾".to_string())?];
    let body_without_comments_and_spaces: String = body
        .lines()
        .filter(|line| !line.trim_start().starts_with("//"))
        .flat_map(str::chars)
        .filter(|character| !character.is_whitespace())
        .collect();
    let expected_body = concat!(
        "self.ledger.take_in_the_devices_kept_before_booking(",
        "|card_position|self.cards[card_position].started.started_now(),",
        "|started:&GpuCard|started.device_condition().is_lost(),",
        "|started:&GpuCard|\u{7b}device_found_idle_by_a_poll(&std::panic::catch_unwind(",
        "std::panic::AssertUnwindSafe(||\u{7b}started.device().poll(wgpu::PollType::Poll)\u{7d}),))\u{7d},",
        "|started:&GpuCard|\u{7b}started.device_on_the_books.as_deref().expect(",
        "\"卡池起的设备起好时放进了账上的那一台（started_card）\")\u{7d},",
        ");"
    );
    if body_without_comments_and_spaces != expected_body {
        return Err("RED: 方法体整句不等于钉住的那一句".to_string());
    }
    Ok(())
}

fn whole_name_uses_of_the_stop_request(line: &str) -> usize {
    let name = concat!("stop_is_", "requested");
    let is_part_of_a_name = |neighbour: Option<char>| {
        neighbour.is_some_and(|neighbour_character| neighbour_character.is_alphanumeric() || neighbour_character == '_')
    };
    line.match_indices(name)
        .filter(|(position, _name)| {
            !is_part_of_a_name(line[..*position].chars().next_back())
                && !is_part_of_a_name(line[position + name.len()..].chars().next())
        })
        .count()
}

fn code_lines_outside_the_tests(source: &str) -> Vec<String> {
    let tests_start = source.find(TESTS_MODULE_OPENING).expect("这份文件有测试模块");
    let tests_end = tests_start + source[tests_start..].find(TOP_LEVEL_CLOSING_BRACE_LINE).expect("测试模块在顶格的右括号处收尾");
    format!("{}{}", &source[..tests_start], &source[tests_end..])
        .lines()
        .filter(|line| !line.trim_start().starts_with("//"))
        .map(str::to_string)
        .collect()
}

fn crash(source: &str) -> Result<(), String> {
    let stop_parameter = concat!("stop_is_", "requested: &(dyn Fn() -> bool + Sync),");
    let handed_on_as_the_last_argument = concat!("stop_is_", "requested,");
    let pinned: Vec<(&str, Vec<&str>)> = vec![
        ("judge_my_blocks_on_gpu_cards", vec![handed_on_as_the_last_argument]),
        (
            "judge_my_blocks_on_gpu_cards_booking_the_scratch_by_where_the_cards_came_from",
            vec![concat!("let stop_request = StopRequestRememberedInThisRun::new(stop_is_", "requested);")],
        ),
        ("check_selected_blocks", vec![handed_on_as_the_last_argument]),
        ("judge_my_blocks_on_gpu", vec![handed_on_as_the_last_argument]),
    ];
    let lines = code_lines_outside_the_tests(source);
    let mut layers = Vec::new();
    for (parameter_line_index, _l) in lines.iter().enumerate().filter(|(_i, line)| line.trim() == stop_parameter) {
        let signature = lines[..parameter_line_index].iter().rev().find(|line| line.contains("fn ")).expect("参数行之前有函数签名");
        let layer = signature.split("fn ").nth(1).and_then(|a| a.split(['(', '<']).next()).expect("fn 之后是函数名").to_string();
        let rest: Vec<&String> = lines[parameter_line_index + 1..].iter().take_while(|line| line.as_str() != "\u{7d}").collect();
        if let Some((_l, expected_lines)) = pinned.iter().find(|(n, _)| *n == layer.as_str()) {
            let lines_using_it: Vec<(usize, &str)> = rest
                .iter()
                .enumerate()
                .filter(|(_o, line)| whole_name_uses_of_the_stop_request(line) > 0)
                .map(|(o, line)| (o, line.trim()))
                .collect();
            let got: Vec<&str> = lines_using_it.iter().map(|(_o, l)| *l).collect();
            if got != *expected_lines {
                return Err(format!("RED: {layer} 用到叫停的行不等于钉住的：{got:?}"));
            }
            for (offset, line) in &lines_using_it {
                if *line == handed_on_as_the_last_argument
                    && !rest.get(offset + 1).is_some_and(|next| next.trim_start().starts_with(')'))
                {
                    return Err(format!("RED: {layer} 交下去的那一行之后不是右括号"));
                }
            }
        }
        layers.push(layer);
    }
    let expected = vec![
        "check_blocks_where_until",
        "check_pending_blocks_in_pieces",
        "judge_my_blocks_on_gpu_cards",
        "judge_my_blocks_on_gpu_cards_booking_the_scratch_by_where_the_cards_came_from",
        "check_selected_blocks",
        "judge_my_blocks_on_gpu",
    ];
    if layers != expected {
        return Err(format!("RED: 层的名单不对 {layers:?}"));
    }
    Ok(())
}

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let mode = &args[1];
    let source = std::fs::read_to_string(&args[2]).expect("读得到");
    let r = match mode.as_str() {
        "gpu" => gpu(&source),
        "crash" => crash(&source),
        other => panic!("{other}"),
    };
    match r {
        Ok(()) => println!("GREEN"),
        Err(e) => println!("{}", if e.starts_with("RED") { e } else { format!("RED(setup): {e}") }),
    }
}
