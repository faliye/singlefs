import itertools, sys

def run_tree(n, advance_at_commit, mkfs_wm, issued):
    total = 0; with_dup = 0; count = 0
    for order in set(itertools.permutations([t for t in range(n) for _ in range(2)])):
        has_file = False; tw = mkfs_wm; step = [0] * n; view = [None] * n; got = [set() for _ in range(n)]
