// 加在 crates/singlefs-harness/src/gpu_memory_ledger.rs 的 mod tests 里、fn room_now 之前（草稿副本里的探针，没进仓）。
// 跑法：cargo test --offline -p singlefs-harness --lib -- --nocapture probe_kept_leases（经 run-with-memory-cap.sh）
    #[test]
    fn probe_kept_leases_on_a_lost_device_block_every_booking_until_a_restart() {
        let (card, ledger) = ledger_on_a_card_under_the_ledger();
        let mut first = ledger.book_on_the_card(0, 1).expect("空卡借得到");
        let device = ledger.record_the_device_started_on_the_card(0);
        {
            let mut card = card.lock().expect("假卡不 panic");
            card.device_started = true;
            card.driver_occupied_mebibytes = 600;
        }
        first.take_in_the_wait_before_returning(
            &device,
            DeviceWaitBeforeReturningTheLease::TheSubmittedWorkFinished,
        );
        drop(first);
        let room_before = room_now(&ledger);
        let big = room_before - 1_000;
        let kept = a_lease_built_and_dropped_after_a_timed_out_wait(&card, &ledger, &device, big);
        // 设备丢了：卡池这一侧知道，账不知道（账只在重起的 prepare 里记）
        {
            let mut card = card.lock().expect("假卡不 panic");
            card.device_started = false;
            card.driver_occupied_mebibytes = 0;
            card.built_mebibytes = 0;
        }
        println!("PROBE room_before={room_before} kept={kept} room_now_after_lost={}", room_now(&ledger));
        let refused = ledger.book_on_the_card(0, 4_000).err();
        println!("PROBE booking_4000_after_the_device_is_lost_and_the_buffers_are_gone={refused:?}");
        assert!(refused.is_some());
        // 卡池只在借到之后才走到起设备；这里模拟重起走到 prepare
        let prepared = ledger.prepare_the_driver_share_before_starting_the_device(0);
        println!("PROBE prepare={prepared:?} room_after_prepare={}", room_now(&ledger));
        assert!(ledger.book_on_the_card(0, 4_000).is_ok());
    }
