#   根环：txg u 落区域 u % 3、区内槽 (u / 3) % S（D22 已定项 1 的 R = 3、S；r2 A2 核它与 u mod 3S 一一对应）；区域设备 [0, 1, 0]。
#   可再分配 = 已释放 ∧ 释放代 ≤ max(F_生效, 环里最旧有效根)（D16 已定项 1）；Hh 里不抬 F ⇒ F_生效 = 0。
#   每一次发布（连空发布）都重写树表单元（D16 已定项 1「非空」段）；mkfs 之后的两次暖机是零单元发布（树表 0 条那一版），不重写。
#   暖机：新实例的根先覆盖两块盘（D16 已定项 8）——写行那次之后接着发，直到两块盘上各有一条本实例的根。
#   c2 崩溃（记录已持久、根槽没持久）之后可写挂载：记录施加，新实例第一个 txg = max(环里最高 txg, 记录最高 txg) + 1（D23 已定项 14 第 3 条）。
# 单元只建树表（每次发布 1 槽、下一次发布换下）：Q1a 的数是真实值的下界；判零的几条（A12、A13）与单元怎么建无关。
import itertools, sys

REGION_DEVICES = [0, 1, 0]
def dev(u): return REGION_DEVICES[u % 3]

def a2(S):
    P = 3 * S
    slots = {(u % 3, (u // 3) % S) for u in range(P)}
    same = all(((u % 3, (u // 3) % S) == ((u + P) % 3, ((u + P) // 3) % S)) for u in range(4 * P))
    return len(slots) == P and same

def build_history(S, k, pos):
    """交回 (events, holes, recovery_ends)：events[u] = 'root' | 'crash'；txg 0..end。"""
    P = 3 * S
    events = {0: 'root', 1: 'root', 2: 'root', 3: 'root'}   # mkfs、两次暖机、第一个事务
    zero_unit = {1, 2}
    holes, recovery_ends, kinds = [], [], {}
    u = 3
    def publish(kind):
        nonlocal u
        u += 1; events[u] = 'root'; kinds[u] = kind
    def crash_and_recover():
        nonlocal u
        u += 1; events[u] = 'crash'; holes.append(u); kinds[u] = 'crash'
        publish('row'); covered = {dev(u)}
        while covered != {0, 1}:
            publish('warm'); covered.add(dev(u))
        recovery_ends.append(u)
    first_hole_at = 5 if pos == 'front' else 3 * S + 4
    while len(holes) < k:
        if u + 1 >= first_hole_at and (not holes or gap >= 2):
            crash_and_recover(); gap = 0
        else:
            publish('work'); gap = (gap + 1) if holes else 0
    return events, holes, recovery_ends, kinds, zero_unit

def run(S, k, pos, end=None):
    P = 3 * S
    events, holes, rec_ends, kinds, zero_unit = build_history(S, k, pos)
    last = max(events)
    target_end = (holes[-1] + P + 2) if holes else None
    if end is None: end = target_end
    u = last
    while u < end:
        u += 1; events[u] = 'root'; kinds[u] = 'work'
    ring = {}
    units = {}          # 树表单元：alloc txg -> release txg（None = 仍分配）；0 是 mkfs 的第 0 版树表
    units[0] = None
    live_tt = 0
    rows = []
    for t in range(0, end + 1):
        if t in zero_unit: pass
        elif t > 0:
            units[live_tt] = t; units[t] = None; live_tt = t
        if events[t] == 'root':
            ring[t % P] = t
        if events[t] != 'root' or t < 3: continue
        oldest = min(ring.values())
        floor = max(0, oldest)
        roots = set(ring.values())
        withheld = {a for a, r in units.items() if r is None or r > floor}
        referenced = {a for a, r in units.items() if r is None or any(a <= x < r for x in roots)}
        g12 = {a for a, r in units.items() if r is None or any(a <= x < r for x in roots)}
        q1a = len(withheld - referenced); q1b = len(g12 - referenced)
        miss = sum(1 for v in range(oldest, t + 1) if v not in roots)
        hdong = sum(1 for h in holes if oldest <= h <= t and h not in roots and ring.get(h % P, -1) < h)
        live = [h for h in holes if h <= t < h + P - (1 if h < P else 0)]
        after = [h for h in live if h >= P]
        hq_formula = (t - min(after) + 1) if after else len(live)
        rows.append(dict(t=t, floor=floor, q1a=q1a, q1b=q1b, h_que=miss, h_dong=hdong,
                         h_dong_formula=len(live), h_que_formula=hq_formula,
                         measured=(t == 3 or kinds.get(t) == 'work' or t in rec_ends)))
    # 同时活着的洞两两不同余 3S
    fits = all(not (a % P == b % P and abs(a - b) < P) for a, b in itertools.combinations(holes, 2)) \
        and all((h - P) not in holes for h in holes if h >= P)
    return rows, holes, rec_ends, fits, end

def a10(S, end):
    P = 3 * S
    lags = set()
    ring = {}
    released = {}
    live_tt = 0
    for t in range(0, end + 1):
        if t not in (1, 2) and t > 0:
            released[live_tt] = t; live_tt = t
        ring[t % P] = t
        floor = min(ring.values())
        for a, r in released.items():
            if r <= floor and (a, r) not in seen:
                seen.add((a, r)); lags.add((t + 1) - r)
    return lags

print("A2", {S: a2(S) for S in (4, 8, 16)})
summary = []
for S in (4, 8, 16):
    P = 3 * S
    ends = {}
    cells = {}
    for pos in ('front', 'back'):
        for k in (1, 2, 4):
            rows, holes, rec, fits, end = run(S, k, pos)
            cells[(k, pos)] = (rows, holes, rec, fits, end)
    k0_end = max(c[4] for c in cells.values())
    rows0, _, _, _, _ = run(S, 0, 'back', end=k0_end)
    seen = set()
    lags = a10(S, k0_end)
    print(f"S={S} A10 lags_first_allocatable_minus_release={sorted(lags)}  k0_end={k0_end}")
    print(f"S={S} A12 k0 max_q1a={max(r['q1a'] for r in rows0)} max_q1b={max(r['q1b'] for r in rows0)} max_h_que={max(r['h_que'] for r in rows0)}")
    for (k, pos), (rows, holes, rec, fits, end) in sorted(cells.items(), key=lambda kv: (kv[0][1], kv[0][0])):
        mm_dong = sum(1 for r in rows if r['h_dong'] != r['h_dong_formula'])
        mm_que = sum(1 for r in rows if r['h_que'] != r['h_que_formula'])
        q1b_max = max(r['q1b'] for r in rows)
        after_rec = [r for r in rows if r['t'] in rec]
        first_after = [r['q1a'] for r in after_rec]
        peak = max(r['q1a'] for r in rows)
        line = (f"S={S} k={k} pos={pos} holes={holes} recovery_ends={rec} end={end} fits={fits} "
                f"A11_mismatch_h_dong={mm_dong} A11_mismatch_h_que={mm_que} A13_max_q1b={q1b_max} "
                f"A16_q1a_at_recovery_ends={first_after} q1a_tt_peak={peak}")
        print(line)
        if pos == 'back' and k == 1:
            t = holes[0]
            fl = {r['t']: r['floor'] for r in rows}
            a14 = all(fl[p] == t - P for p in range(t + 1, t + P) if p in fl) and fl.get(t + P) == t + 1
            q = {r['t']: r['q1a'] for r in rows}
            print(f"S={S} A14 hole={t} floor_pinned_to={t-P} over_[{t+1},{t+P-1}] then_{t+1}_at_{t+P}: {a14}")
            print(f"S={S} A15 q1a_tt_at_t+3S-1={q[t+P-1]}  equals_3S={q[t+P-1]==P}")

if '--dump' in sys.argv:
    # 逐观测点：装置在每次它自己驱动的工作负载发布之后、每次可写挂载做完之后各观测一次（写行与暖机在挂载里面，看不到中间态）
    for S in (4, 8, 16):
        cells = {(k, pos): run(S, k, pos) for pos in ('front', 'back') for k in (1, 2, 4)}
        k0_end = max(c[4] for c in cells.values())
        cells[(0, '-')] = run(S, 0, 'back', end=k0_end)
        for (k, pos), (rows, holes, rec, fits, end) in sorted(cells.items(), key=lambda kv: (kv[0][1], kv[0][0])):
            for r in rows:
